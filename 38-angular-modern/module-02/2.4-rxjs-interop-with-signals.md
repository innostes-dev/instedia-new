# 2.4 RxJS Interop with Signals

## Phase A: Concept & Application

### 1. Real-World Pain Point (The Why)
RxJS is incredibly powerful for handling asynchronous events over time (like debouncing keystrokes, canceling HTTP requests, or handling WebSockets). However, historically, binding RxJS Observables directly to the UI required the `| async` pipe. This pipe had major drawbacks: it forced you to handle `null` states explicitly (`*ngIf="data$ | async as data"`), it caused multiple subscriptions if used repeatedly in the same template, and it still relied heavily on Zone.js to notify Angular when the Observable emitted a new value. In complex apps like a Live Reactive Search Autocomplete, managing these subscriptions and avoiding memory leaks was a significant source of bugs.

### 2. Plain English Metaphor (The What)
Imagine RxJS as a raging river (a stream of data over time, like user keystrokes).
Imagine Signals as a calm reservoir lake (a snapshot of the current state).
The `| async` pipe was like drinking directly from the raging river with a bucket. It was messy and required constant attention.
`toSignal()` is a water treatment plant. It connects to the river (subscribes to the RxJS Observable), continuously fills the calm reservoir (the Signal) with the latest clean water, and automatically shuts off the intake (unsubscribes) when the reservoir is demolished (component destroyed). The UI then simply drinks safely from the calm lake.

### 3. Complete Runnable Code
Here is a Live Reactive Search Autocomplete with Debounce & SwitchMap, converting between RxJS and Signals.

```typescript
import { Component, signal, effect } from '@angular/core';
import { toObservable, toSignal } from '@angular/core/rxjs-interop';
import { Observable, of } from 'rxjs';
import { debounceTime, distinctUntilChanged, switchMap, catchError, startWith } from 'rxjs/operators';
import { HttpClient } from '@angular/common/http';

interface SearchResult { id: number; title: string; type: string; }

@Component({
  selector: 'app-search-autocomplete',
  standalone: true,
  template: `
    <div class="search-container">
      <h2>Live Database Search</h2>

      <div class="input-wrapper">
        <input
          type="text"
          placeholder="Search for users or documents..."
          (input)="updateSearch($event)"
          [value]="searchTerm()"
          class="search-input">

        <!-- UI responds to Signals, completely avoiding | async pipes -->
        @if (isLoading()) {
          <div class="spinner"></div>
        }
      </div>

      <div class="results-panel">
        <!-- We read the RxJS output as a simple Signal -->
        @for (result of results(); track result.id) {
          <div class="result-item">
            <span class="type-badge" [class]="result.type">{{ result.type }}</span>
            <span class="title">{{ result.title }}</span>
          </div>
        } @empty {
           <!-- Handle different empty states -->
           @if (searchTerm().length === 0) {
             <div class="empty-msg">Type at least 2 characters to search...</div>
           } @else if (!isLoading()) {
             <div class="empty-msg">No results found for "{{ searchTerm() }}"</div>
           }
        }
      </div>
    </div>
  `,
  styles: [`
    .search-container { max-width: 500px; margin: 2rem auto; font-family: sans-serif; }
    .input-wrapper { position: relative; display: flex; align-items: center; }
    .search-input { width: 100%; padding: 1rem; font-size: 1.1rem; border: 2px solid #cbd5e1; border-radius: 8px; outline: none; transition: border-color 0.2s; }
    .search-input:focus { border-color: #3b82f6; }
    .spinner { position: absolute; right: 1rem; width: 20px; height: 20px; border: 3px solid #f3f3f3; border-top: 3px solid #3b82f6; border-radius: 50%; animation: spin 1s linear infinite; }
    @keyframes spin { 0% { transform: rotate(0deg); } 100% { transform: rotate(360deg); } }
    .results-panel { margin-top: 0.5rem; background: white; border: 1px solid #e2e8f0; border-radius: 8px; box-shadow: 0 4px 6px -1px rgba(0,0,0,0.1); overflow: hidden; }
    .result-item { padding: 1rem; border-bottom: 1px solid #f1f5f9; display: flex; align-items: center; gap: 1rem; cursor: pointer; transition: background 0.2s; }
    .result-item:hover { background: #f8fafc; }
    .type-badge { font-size: 0.7rem; padding: 0.2rem 0.5rem; border-radius: 4px; font-weight: bold; text-transform: uppercase; }
    .type-badge.user { background: #dbeafe; color: #1e40af; }
    .type-badge.doc { background: #fce7f3; color: #9d174d; }
    .empty-msg { padding: 2rem; text-align: center; color: #64748b; font-style: italic; }
  `]
})
export class SearchAutocompleteComponent {
  // 1. UI State (Signals)
  searchTerm = signal<string>('');
  isLoading = signal<boolean>(false);

  // 2. RxJS Stream: Convert Signal -> Observable
  // We do this because RxJS is better at time-based operations like debounce and cancel (switchMap)
  searchTerm$ = toObservable(this.searchTerm);

  // 3. RxJS Pipeline
  results$ = this.searchTerm$.pipe(
    debounceTime(300), // Wait 300ms after last keystroke
    distinctUntilChanged(), // Don't search if the term hasn't actually changed
    switchMap(term => {
      if (term.length < 2) {
        this.isLoading.set(false);
        return of([]); // Return empty array if term is too short
      }

      this.isLoading.set(true);
      // Mock HTTP call. In real app: return this.http.get(...)
      return this.mockApiCall(term).pipe(
        catchError(() => of([])), // Handle errors gracefully
      );
    })
  );

  // 4. Back to UI: Convert Observable -> Signal
  // The UI binds exclusively to this signal. No | async pipe needed.
  // We provide an initial value so we don't have to deal with undefined in the template.
  results = toSignal(this.results$, { initialValue: [] as SearchResult[] });

  constructor() {
    // We can use an effect to turn off the loading spinner when results arrive
    effect(() => {
      // Whenever results() updates, we know the API call finished
      const res = this.results();
      this.isLoading.set(false);
    }, { allowSignalWrites: true }); // Note: generally avoid writes in effects, but safe here for simple UI state
  }

  updateSearch(event: Event) {
    this.searchTerm.set((event.target as HTMLInputElement).value);
  }

  // --- Mock Backend ---
  private mockApiCall(term: string): Observable<SearchResult[]> {
    return new Observable(subscriber => {
      const db: SearchResult[] = [
        { id: 1, title: 'John Doe', type: 'user' },
        { id: 2, title: 'Jane Smith', type: 'user' },
        { id: 3, title: 'Q3 Financial Report.pdf', type: 'doc' },
        { id: 4, title: 'Architecture Diagram.png', type: 'doc' },
      ];

      const filtered = db.filter(item =>
        item.title.toLowerCase().includes(term.toLowerCase())
      );

      // Simulate network latency
      setTimeout(() => {
        subscriber.next(filtered);
        subscriber.complete();
      }, 800);
    });
  }
}
```

### 4. Progressive Step-by-Step Breakdown

**Step 1: Convert a Signal to an Observable using `toObservable()`**
*Rationale: Signals are great for representing "right now", but terrible at representing "wait 300ms" (Debounce). By converting the `searchTerm` signal to an RxJS Observable, we unlock the entire arsenal of RxJS time-based operators.*
```typescript
import { toObservable } from '@angular/core/rxjs-interop';

searchTerm = signal('');
// Creates an Observable that emits every time the signal changes
searchTerm$ = toObservable(this.searchTerm);
```

**Step 2: Process the stream with RxJS**
*Rationale: We pipe the stream through `debounceTime` and `switchMap`. `switchMap` is critical for typeaheads because if the user types "a", triggers a request, and then types "ab", `switchMap` automatically cancels the first pending HTTP request so we don't get out-of-order race conditions.*
```typescript
results$ = this.searchTerm$.pipe(
  debounceTime(300),
  switchMap(term => this.http.get(`/api/search?q=${term}`))
);
```

**Step 3: Convert the Observable back to a Signal using `toSignal()`**
*Rationale: Instead of using `*ngIf="results$ | async as results"`, we use `toSignal`. It subscribes to the Observable immediately. When the component is destroyed, it automatically unsubscribes, preventing memory leaks. We provide `initialValue: []` so the template doesn't crash trying to render `undefined` before the first HTTP request completes.*
```typescript
import { toSignal } from '@angular/core/rxjs-interop';

results = toSignal(this.results$, { initialValue: [] });
```

### 5. ASCII / Architecture Data Flow
```
[User Types "J"] -> this.searchTerm.set("J") -> [UI Updates instantly]
       |
[toObservable]
       |
(Wait 300ms) --[DebounceTime]--> (RxJS Pipeline) --[SwitchMap (HTTP)]
                                                         |
                                                    [Server Response]
                                                         |
[toSignal] <---------------------------------------------+
       |
this.results() signal updates with data -> [UI Renders List instantly]
```

### 6. Noob Trap vs Pro Tip
**Noob Trap:** Using RxJS for derived state (e.g., `total$ = combineLatest([price$, tax$]).pipe(map(...))`). While this works, it is unnecessarily complex.
**Pro Tip:** Use Signals and `computed()` for all synchronous, derived state (math, filtering arrays locally). Use RxJS *only* for asynchronous events, race conditions, and time-based orchestration (HTTP calls, WebSockets, debouncing, polling). The boundary between them should be HTTP requests.

### Level 3 Boss Level (Cumulative)
**Task:** Combine Signals (2.1), Inputs (2.3), and RxJS Interop (2.4).
Build a `StockPriceWidget` component.
1. It takes an `input.required<string>()` called `ticker` (e.g., "AAPL").
2. Convert the `ticker` signal to an observable using `toObservable`.
3. Pipe it through `switchMap` to a mock API call that returns a price for that ticker.
4. Convert the output stream back to a signal `price`.
5. Display the ticker symbol and the price in the template. If the parent changes the `ticker` input, the child should automatically fetch the new price.

<details>
<summary>💡 Hint</summary>
Because `ticker` is an input signal, changing it triggers `toObservable(this.ticker)`, which flows into `switchMap`, fetching the new data automatically!
</details>

<details>
<summary>✅ Solution Walkthrough</summary>

```typescript
import { Component, input, signal } from '@angular/core';
import { toObservable, toSignal } from '@angular/core/rxjs-interop';
import { of, delay } from 'rxjs';
import { switchMap, tap } from 'rxjs/operators';

@Component({
  selector: 'app-stock-widget',
  standalone: true,
  template: `
    <div style="border: 1px solid #ccc; padding: 1rem; border-radius: 8px; width: 200px;">
      <h2>{{ ticker() }}</h2>
      @if (loading()) {
        <p>Fetching price...</p>
      } @else {
        <p style="font-size: 1.5rem; font-weight: bold;">{{ price() | currency }}</p>
      }
    </div>
  `
})
export class StockWidgetComponent {
  ticker = input.required<string>();
  loading = signal(false);

  // 1. Observe the input signal
  ticker$ = toObservable(this.ticker);

  // 2. RxJS logic to fetch data when input changes
  price$ = this.ticker$.pipe(
    tap(() => this.loading.set(true)),
    switchMap(symbol => this.mockApi(symbol)),
    tap(() => this.loading.set(false))
  );

  // 3. Back to Signal for the UI
  price = toSignal(this.price$, { initialValue: 0 });

  private mockApi(symbol: string) {
    // Generate random price based on length for consistency in mock
    const randomPrice = symbol.length * 50 + (Math.random() * 20);
    return of(randomPrice).pipe(delay(800)); // Simulate network
  }
}

// Parent Usage
@Component({
  selector: 'app-portfolio',
  standalone: true,
  imports: [StockWidgetComponent],
  template: `
    <button (click)="selectedSymbol.set('AAPL')">Apple</button>
    <button (click)="selectedSymbol.set('TSLA')">Tesla</button>
    <button (click)="selectedSymbol.set('MSFT')">Microsoft</button>

    <br><br>
    <!-- Changing the signal automatically triggers the child's RxJS stream! -->
    <app-stock-widget [ticker]="selectedSymbol()" />
  `
})
export class PortfolioComponent {
  selectedSymbol = signal('AAPL');
}
```
